## 1. Visión General del Proyecto

### 1.1 Descripción

Sistema completo de gestión de tareas estilo Jira/Trello/Azure DevOps. Permite a equipos organizar proyectos, crear y asignar tareas, rastrear progreso, comentar, adjuntar archivos y mantener un historial completo de auditoría.

### 1.2 Stack Tecnológico

| Capa | Tecnología |
|------|-----------|
| Frontend | Angular 20, Standalone Components, Signals, PrimeNG, TailwindCSS, RxJS, Reactive Forms |
| Backend | .NET 8, ASP.NET Core Web API, MediatR, CQRS, FluentValidation, Entity Framework Core, Mapster |
| Auth | JWT + Refresh Tokens, ASP.NET Core Identity |
| BD | SQL Server Local |
| Logs | Serilog (Console + File) |
| Dev | Swagger/OpenAPI (solo desarrollo local) |

### 1.3 Principios Rectores

- **Clean Architecture** — Separación estricta por dependencias (Domain → Application → Infrastructure → API)
- **SOLID** — Cada clase tiene una única responsabilidad, dependencias invertidas
- **CQRS** — Commands (escritura) separados de Queries (lectura)
- **DRY / KISS** — No duplicar lógica, simplicidad sobre complejidad
- **Convention over Configuration** — Convenciones claras que reducen decisiones

### 2.2 Diagrama de Dependencias (Clean Architecture)

```
┌─────────────────────────────────────────────────────┐
│                  API (Web)                           │
│         Controllers, Middleware, Filters             │
│                  ┌───────┐                           │
│                  │       │                           │
│                  ▼       ▼                           │
│           ┌──────────────────────┐                   │
│           │     Application      │ ← CQRS Handlers   │
│           │  Commands, Queries   │   Validators      │
│           │  Behaviors, DTOs     │   Business Rules  │
│           └──────────┬───────────┘                   │
│                      │                               │
│              ┌───────┴───────┐                       │
│              ▼               ▼                       │
│     ┌──────────────┐  ┌─────────────┐                │
│     │   Domain     │  │ Shared      │                │
│     │  Entities    │  │ DTOs, Const │                │
│     │  Enums       │  │ Interfaces  │                │
│     │  Exceptions  │  │ Exceptions  │                │
│     └──────────────┘  └─────────────┘                │
│              ▲               ▲                       │
│              │               │                       │
│     ┌────────┴───────────────┴────────┐              │
│     │      Infrastructure             │              │
│     │  EF Core, Repositories          │              │
│     │  JWT, FileStorage, Services     │              │
│     └─────────────────────────────────┘              │
└─────────────────────────────────────────────────────┘
```

**Regla de dependencia:** Las flechas SOLO apuntan hacia adentro. Domain nunca depende de Application. Application nunca depende de Infrastructure. API es el único proyecto que referencia a todos.

## 4. Backend — API y Dominio

### 4.1 Configuración de Dependency Injection

```csharp
// TaskManagement.Infrastructure/DependencyInjection.cs
public static class DependencyInjection
{
    public static IServiceCollection AddInfrastructure(
        this IServiceCollection services, IConfiguration config)
    {
        // DbContext
        services.AddDbContext<AppDbContext>(options =>
            options.UseSqlServer(config.GetConnectionString("DefaultConnection")));

        // Identity
        services.AddIdentityCore<User>(options => { ... })
            .AddRoles<Role>()
            .AddEntityFrameworkStores<AppDbContext>();

        // JWT
        services.Configure<JwtSettings>(config.GetSection("JwtSettings"));
        services.AddScoped<IJwtTokenService, JwtTokenService>();
        services.AddScoped<IRefreshTokenService, RefreshTokenService>();

        // Repositories
        services.AddScoped<IUnitOfWork, UnitOfWork>();

public record CreateTaskCommand : IRequest<BaseResponse<Guid>>
{

    public Guid ProjectId { get; init; }
    public Guid? AssignedToId { get; init; }

    public DateTime? DueDate { get; init; }
    public List<string>? Tags { get; init; }
}

// Validator
public class CreateTaskValidator : AbstractValidator<CreateTaskCommand>
{
    public CreateTaskValidator()
    {
        RuleFor(x => x.Title)
            .NotEmpty().WithMessage("Title is required")
            .MaximumLength(200).WithMessage("Title must not exceed 200 characters");

        RuleFor(x => x.ProjectId)
            .NotEmpty().WithMessage("Project is required");
    }
}

// Handler
public class CreateTaskHandler
    : IRequestHandler<CreateTaskCommand, BaseResponse<Guid>>
{
    private readonly IUnitOfWork _unitOfWork;
    private readonly ICurrentUserService _currentUser;

    public CreateTaskHandler(IUnitOfWork unitOfWork, ICurrentUserService currentUser)
    {
        _unitOfWork = unitOfWork;
        _currentUser = currentUser;
    }

    public async Task<BaseResponse<Guid>> Handle(
        CreateTaskCommand request, CancellationToken ct)
    {
        var task = new TaskItem
        {
            Title = request.Title,
            Description = request.Description,
            ProjectId = request.ProjectId,
            AssignedToId = request.AssignedToId,
            CreatedById = _currentUser.UserId,
            Priority = (TaskPriority)request.Priority,
            DueDate = request.DueDate,
            Tags = request.Tags != null ? string.Join(",", request.Tags) : null
        };

        await _unitOfWork.Tasks.AddAsync(task, ct);
        await _unitOfWork.SaveChangesAsync(ct);

        return BaseResponse<Guid>.Success(task.Id);
    }
}

public class BaseResponse<T>
{
    public bool Success { get; set; }
    public string? Message { get; set; }
    public T? Data { get; set; }
    public List<string>? Errors { get; set; }

    public static BaseResponse<T> Success(T data, string? message = null) => ...
    public static BaseResponse<T> Failure(List<string> errors) => ...
}

public class PaginatedList<T>
{
    public List<T> Items { get; set; }
    public int TotalCount { get; set; }
    public int Page { get; set; }
    public int PageSize { get; set; }
    public int TotalPages => (int)Math.Ceiling(TotalCount / (double)PageSize);
    public bool HasPrevious => Page > 1;
    public bool HasNext => Page < TotalPages;
}

@Injectable({ providedIn: 'root' })
export class AuthStore {
  private authService = inject(AuthService);

  // State (Signals)
  private _user = signal<User | null>(null);
  private _isAuthenticated = signal<boolean>(false);
  private _isLoading = signal<boolean>(false);

  // Computed
  user = this._user.asReadonly();
  isAuthenticated = this._isAuthenticated.asReadonly();
  isLoading = this._isLoading.asReadonly();
  isAdmin = computed(() => this._user()?.roles?.includes('Admin') ?? false);

  // Actions
  async login(credentials: LoginRequest): Promise<void> {
    this._isLoading.set(true);
    try {
      const response = await firstValueFrom(this.authService.login(credentials));
      this._user.set(response.data.user);
      this._isAuthenticated.set(true);
      localStorage.setItem('accessToken', response.data.accessToken);
    } finally {
      this._isLoading.set(false);
    }
  }

  logout(): void {
    this._user.set(null);
    this._isAuthenticated.set(false);
    localStorage.removeItem('accessToken');
    localStorage.removeItem('refreshToken');
  }
}
```

// error.interceptor.ts
export const errorInterceptor: HttpInterceptorFn = (req, next) => {
  const router = inject(Router);
  const toast = inject(ToastService);

  return next(req).pipe(
    catchError((error: HttpErrorResponse) => {
      if (error.status === 401) {
        localStorage.clear();
        router.navigate(['/auth/login']);
      } else if (error.status === 403) {
        toast.error('No tienes permisos para esta acción');
      } else if (error.status === 404) {
        toast.error('Recurso no encontrado');
      } else if (error.status >= 500) {
        toast.error('Error del servidor. Intenta de nuevo.');
      }
      return throwError(() => error);
    })
  );
};

### 6.2 Flujo de Refresh Automático (Frontend)

```typescript
// Interceptor que renueva el token automáticamente
export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const authService = inject(AuthService);
  const token = localStorage.getItem('accessToken');

  if (!token) return next(req);

  // Verificar si el token está por expirar (5 min antes)
  const payload = JSON.parse(atob(token.split('.')[1]));
  const expiresAt = payload.exp * 1000;
  const now = Date.now();

  if (expiresAt - now < 5 * 60 * 1000) {
    // Token por expirar, renovar
    return authService.refreshToken().pipe(
      switchMap(newToken => {
        const cloned = req.clone({
          setHeaders: { Authorization: `Bearer ${newToken}` }
        });
        return next(cloned);
      })
    );
  }

  return next(req.clone({
    setHeaders: { Authorization: `Bearer ${token}` }
  }));
};
```
